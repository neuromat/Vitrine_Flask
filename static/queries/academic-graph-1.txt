SELECT ?Autores (COUNT(?artigo) AS ?quantidade_artigos)
WHERE {
  {
    SELECT ?artigo (COUNT(DISTINCT ?autor) AS ?Autores)
    WHERE {
      # 1. Identifica os artigos da coleção
      ?artigo wdt:P361 wd:Q18477654 .

      # 2. Identifica os autores desses artigos
      ?artigo wdt:P50 ?autor .

      # 3. Filtro de Segurança: Garante que o autor tenha vínculo com a coleção
      # Foi usada uma subquery simples para validar o autor, o que é mais rápido que um JOIN de obras.
      FILTER EXISTS {
        ?any_obra wdt:P361 wd:Q18477654 ;
                  wdt:P50 ?autor .
      }
    }
    GROUP BY ?artigo
    HAVING(?Autores > 1)
  }
}
GROUP BY ?Autores
ORDER BY ?Autores
