# Conta pesquisadores com grau de mestre que são principais e associados
## NÃO ESTÁ SENDO USADO NA RENDERIZAÇÃO ##
## MANTIDA SOMENTE PARA CASO SEJA NECESSÁRIA EM ALGUM MOMENTO INSERIR OUTRA QUERY NESSE LOCAL ##
SELECT (COUNT(?pesquisador) AS ?valor) WHERE {
  ?artigos wdt:P361 wd:Q18477654 ;
           wdt:P50 ?pesquisador.
  ?pesquisador wdt:P106 wd:Q1650915.
  FILTER EXISTS { ?pesquisador wdt:P463 wd:Q18477654. }    # principais
  FILTER EXISTS { ?pesquisador wdt:P1416 wd:Q18477654. }   # associados
  FILTER EXISTS { ?pesquisador wdt:P185 ?d. }              # com orientador (proxy para mestre)
}
