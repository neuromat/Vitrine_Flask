SELECT ?assuntoLabel WHERE {
  # LADO SCHOLARLY
  ?artigo wdt:P361 wd:Q18477654 ;
          wdt:P921 ?assunto .

  # LADO WIKIDATA PRINCIPAL (busca os nomes)
  SERVICE <https://query.wikidata.org/sparql> {
    OPTIONAL { ?assunto rdfs:label ?lPT . FILTER(LANG(?lPT) = "pt") }
    OPTIONAL { ?assunto rdfs:label ?lEN . FILTER(LANG(?lEN) = "en") }
  }

  BIND(COALESCE(?lPT, ?lEN, STR(?assunto)) AS ?assuntoLabel)
}
GROUP BY ?assuntoLabel
ORDER BY DESC(COUNT(?artigo))
LIMIT 10
